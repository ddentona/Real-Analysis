\section{Problem 5.3}
    For each of the following, provide a proof or a  counterexample. 
    \begin{enumerate}[label=(\alph*)]
        \item   If $A$ and $B$ are compact, must $A \cup B$ be compact?
        \item   If $A$ and $B$ are compact, must $A \cap B$ be compact?
    \end{enumerate}

    \subsection{Solution (a)}
        \begin{proof}
            By the Heine-Borel Theorem, if $A$ and $B$ are compact, the two of them are closed and bounded.
            We can name bounds for the two of these and recall the definition of being closed (having an open complement).
            \begin{gather}
                \forall a \in A, L_A \leq a \leq U_A\\
                \forall b \in B, L_B \leq b \leq U_B\\
                \forall x \in A^c, \exists \delta > 0 : (x - \delta, x + \delta) \subseteq A^c\\
                \forall x \in B^c, \exists \delta > 0 : (x - \delta, x + \delta) \subseteq B^c
            \end{gather}

            Suppose we define the union of $A$ and $B$ and its complement by DeMorgan's Law.
            \begin{gather}
                A \cup B = \left\{ c \vert c \in A \vee c \in B \right\}\\
                (A \cup B)^c = A^c \cap B^c
            \end{gather}

            Since both $A^c$ and $B^c$ are open and the intersection of a finite number of open sets is also open, $A^c \cap B^c$ is open.
            By extension, $(A \cup B)^c$ is open and $A \cup B$ is closed.

            Suppose that $U = \max(U_A, U_B)$ and $L = \min(L_A, L_B)$.
            \begin{gather}
                \forall a \in A, U \geq U_A \geq a\\
                \forall b \in B, U \geq U_B \geq b\\
                \forall a \in A, L \leq L_A \leq a\\
                \forall b \in B, L \leq L_B \leq b
            \end{gather}

            From this, we conclude that $U$ is an upper bound of $A \cup B$ and $L$ is a lower bound of $A \cup B$.
            This means that $A \cup B$ is bounded.
            Since $A \cup B$ is both closed and bounded, then by the Heine-Borel Theorem, it is compact.
        \end{proof}

    \subsection{Solution (b)}
        \begin{proof}
            By the Heine-Borel Theorem, if $A$ and $B$ are compact, the two of them are closed and bounded.
            This means that $A^c$ and $B^c$ are open.
            This means that $A^c \cup B^c$, which can be turned into $(A \cap B)^c$ by DeMorgan's Laws, is open.
            $A \cap B$ is closed.

            We can write the definition of $A \cap B$ and name bounds for the two of these.
            \begin{gather}
                A \cap B = \left\{ c \vert c \in A \andtxt c \in B \right\}\\
                \forall a \in A, L_A \leq a \leq U_A\\
                \forall b \in B, L_B \leq b \leq U_B\\
                \forall c \in A \cap B, L_A \leq c \leq U_A \andtxt L_B \leq c \leq U_B
            \end{gather}

            Since there is a minimum and maximum bound of $c$, we can say that $A \cap B$ is bounded.

\section{Problem 5.21}
    For sets $A$ and $B$, define $A \cdot B = {a \cdot b: a \in A, b \in B}$.  
    \begin{enumerate}[label=(\alph*)]
        \item   If $A$ and $B$ are open, must $A \cdot B$ be open?
        \item   If $A$ and $B$ are compact, must $A \cdot B$ be compact?
        \item   If $A$ and $B$ are closed, must $A \cdot B$ be closed?
    \end{enumerate}

    \subsection{Solution (a)}
        Yes.
        \begin{proof}
            Recall what it means that $A$ and $B$ are open.
            For $A$, it means that for all $a \in A$, there exists some $\delta_a > 0$ such that $(a - \delta_a, a + \delta_a) \subseteq A$.
            For $B$, it means that for all $b \in B$, there exists some $\delta_b > 0$ such that $(b - \delta_b, b + \delta_b) \subseteq B$.

            Suppose some term $a \in A$ and some term $b \in B$.
            We wind up with multiple cases, based on whether $a$ or $b$ or both is positive or negative, or if either is zero.
            \begin{case}{1}{Both positive}
                If both are positive, that means that $\frac{\delta_b}{a} > 0$.
                \begin{equation}
                    \beta \in \left(b - \frac{\delta_b}{a}, b + \frac{\delta_b}{a}\right) \subseteq B
                \end{equation}

                This can be rewritten, for all $\beta$.
                \begin{gather}
                    b - \frac{\delta_b}{a} < \beta < b + \frac{\delta_b}{a}
                \end{gather}

                Multiply all sides by $a$.
                Since $a \in A$ and $\beta \in B$, this means that $a\beta \in A \cdot B$.
                \begin{gather}
                    ab - \delta_b < a\beta < ab + \delta_b
                \end{gather}

                This means that $\forall a\beta \in \left( ab - \delta_b, ab + \delta_b \right)$, $a\beta \in A \cdot B$.
                It also means that $\left( ab - \delta_b, ab + \delta_b \right) \subseteq A \cdot B$.
                This means that $A \cdot B$ is open.
            \end{case}

            \begin{case}{2}{Either or both negative}
                Suppose that just $a$ is negative.
                If only $b$ is negative, this can be changed appropriately.
                This means that $\frac{\delta_b}{a} < 0$, and that $-\frac{\delta_b}{a} > 0$.
                \begin{equation}
                    \beta \in \left(b + \frac{\delta_b}{a}, b - \frac{\delta_b}{a}\right) \subseteq B
                \end{equation}

                This can be rewritten, for all $\beta$.
                \begin{gather}
                    b + \frac{\delta_b}{a} < \beta < b - \frac{\delta_b}{a}
                \end{gather}
                
                Multiply all sides by $a$.
                This will reverse the signs.
                Since $a \in A$ and $\beta \in B$, this means that $a\beta \in A \cdot B$.
                \begin{gather}
                    ab + \delta_b > a\beta > ab - \delta_b
                \end{gather}

                This means that $\forall a\beta \in \left( ab - \delta_b, ab + \delta_b \right)$, $a\beta \in A \cdot B$.
                It also means that $\left( ab - \delta_b, ab + \delta_b \right) \subseteq A \cdot B$.
            \end{case}

            \begin{case}{3}{Either are zero}
                Suppose that $a = 0$ and $b \neq 0$.
                We can rewrite our open interval.
                \begin{equation}
                    \left(a - \frac{\delta_a}{b}, a + \frac{\delta_a}{b}\right) = \left(-\frac{\delta_a}{b}, \frac{\delta_a}{b}\right) \subseteq A
                \end{equation}

                Multiply it by $b \in B$.
                \begin{gather}
                    \left(-\delta_a, \delta_a\right) \subseteq A \cdot B
                \end{gather}
            \end{case}

            \begin{case}{4}{Both are zero}
                Suppose that $a = 0$ and $b = 0$.
                Recall that $\delta_b > 0$, which also means $\frac{\delta_b}{2} > 0$.
                We can rewrite our open interval.
                \begin{gather}
                    \left(a - \frac{\delta_a}{\delta_b}, a + \frac{\delta_a}{\delta_b}\right) = \left(-\frac{\delta_a}{\delta_b}, \frac{\delta_a}{\delta_b}\right) \subseteq A
                \end{gather}

                Multiply it by $b + \frac{\delta_b}{2} = \frac{\delta_b}{2} \in B$.
                \begin{gather}
                    \left(-\frac{\delta_a}{2}, \frac{\delta_a}{2}\right) \subseteq A \cdot B
                \end{gather}
            \end{case}

            This covers all the cases, which means that $A \cdot B$ is open.
        \end{proof}

    \subsection{Solution (b)}
        Because $A$ and $B$ are compact, both are closed and bounded.
        This means that all $a \in A$ and $b \in B$ are within intervals.
        \begin{gather}
            L_A < a < U_A\\
            L_B < b < U_B
        \end{gather}

        We can multiply the terms together, and that establish a set of configurations depending on which terms are negative.
        \begin{gather}
            S = \{L_A \cdot L_B, U_A \cdot L_B, L_A \cdot U_B, U_A \cdot U_B\}\\
            \min\{S\} < ab < \max\{S\}
        \end{gather}
    
        This means $A \cdot B$ is bounded.

        Suppose now that there is a sequence in $A \cdot B$ called \seq{x}.
        For all elements in \seq{x}, each element is a multiple of some element $a_n$ and another $b_n$.
        \begin{gather}
            x_n = a_n b_n
        \end{gather}

        For every $a \in A$ and $b \in B$, there exists a subsequence of \seq{a} and \seq{b} that converges to $a$ and $b$ respectively.
        \begin{gather}
            a_{n_k} \to a;
            b_{n_j} \to b
        \end{gather}

        Since $a \in A$ and $b \in B$, $ab \in A \cdot B$.
        Due to the sequence limit laws, $a_{n_k}b_{n_j}$ converges to $a \cdot b$.
        This means that for any $a \cdot b \in A \cdot B$, there is a subsequence \seq[n_j]{x} of \seq{x} that converges to $a \cdot b$.
        Therefore, $A \cdot B$ is compact.

        This means that $A \cdot B$ does not contain all its limit points, so it is not closed.

\section{Problem 5.22}
    Let $A$ be the set of numbers in $\left[ 0, 1 \right]$ whose decimal expansions use only the numbers $2$, $5$ and $8$. 
    For example, $0.5822585582 \in A$, and $0.2525252525\ldots \in A$. 
    Prove that $A$ is a closed set.
    
    \subsection{Solution}
    \begin{proof}[Proof by Contradiction]
        Suppose for contradiction that a sequence in $A$ converges to a point $a \notin A$.
        Throughout, use the decimal expansion that does not end in an infinite string of 9's.
        \begin{gather}
            \forall \varepsilon > 0, \exists N : \forall n > N, \abs{a_n - a} < \varepsilon
        \end{gather}

        $a$ and every $a_n$ can be written in the form of a sum of numbers.
        \begin{gather}
            a_n = \sum_{k=1}^{n} s_k \times 10^{-k}, s_k \in \{2, 5, 8\}\\
            a = \sum_{k=1}^{\infty} t_k \times 10^{-k}, t_k \in \{2, 5, 8, m\}, m \in \{1, 3, 4, 6, 7, 9, 0\}
        \end{gather}

        We can put these into the convergence equation.
        \begin{gather}
            s_k \in \{2, 5, 8\}; t_k \in \{2, 5, 8, m\}; m \in \{1, 3, 4, 6, 7, 9, 0\}\\
            \abs{\sum_{k=1}^{n} s_k \times 10^{-k} - \sum_{k=1}^{\infty} t_k \times 10^{-k}} < \varepsilon\\
            \abs{\sum_{k=1}^{n} (s_k - t_k) \times 10^{-k} - \sum_{k=n+1}^{\infty} t_k \times 10^{-k}} < \varepsilon
        \end{gather}

        Suppose that point $j$ is the first point where $t_j \in m$ and $s_k \neq t_k$, while $s_j \in \{2, 5, 8\}$.
        We can simplify $\sum_{k=1}^{n} (s_k - t_k) \times 10^{-k}$.
        \begin{gather}
            \sum_{k=1}^{n} (s_k - t_k) \times 10^{-k} = \sum_{k=j}^{n} (s_k - t_k) \times 10^{-k}\\
            \abs{\sum_{k=j}^{n} (s_k - t_k) \times 10^{-k} - \sum_{k=n+1}^{\infty} t_k \times 10^{-k}} < \varepsilon
        \end{gather}

        At point $j$, $\abs{s_j - t_j} \geq 1$, which means that $s_j - t_j \geq 1$ or $t_j - s_j \geq 1$.
        This gives us two cases.

        \begin{case}{1}{$s_j - t_j \geq 1$}
            Take the above convergence equation.
            If $n > j$, then the maximum value of $s_k - t_k$ is $8$.
            Even if for all $k \geq n+1$, $t_k = 9$, it will always be the case that the magnitude of the sum up to and including $k = n$ will be greater than the sum after that point.
            \begin{gather}
                \sum_{k=j}^{n} (s_k - t_k) \times 10^{-k} > \sum_{k=n+1}^{\infty} t_k \times 10^{-k}\\
                \abs{\sum_{k=j}^{n} (s_k - t_k) \times 10^{-k} - \sum_{k=n+1}^{\infty} t_k \times 10^{-k}} > 0\\
                \sum_{k=j}^{n} (s_k - t_k) \times 10^{-k} - \sum_{k=n+1}^{\infty} t_k \times 10^{-k} < \varepsilon
            \end{gather}

            Our sum of sums will always be greater than or equal to $10^{-j}$.
            \begin{gather}
                10^{-j} \leq \sum_{k=j}^{n} (s_k - t_k) \times 10^{-k} - \sum_{k=n+1}^{\infty} t_k \times 10^{-k} < \varepsilon
            \end{gather}

            This creates a contradiction, since $\varepsilon$ should be able to be less than $10^{-j}$.

            There is one edge case, when $j = n$, $s_j - t_j = 1$, and $t_k = 9$ for all $k > n$.
            In that case, $\sum_{k=n+1}^{\infty} t_k \times 10^{-k} = 0.\overline{99} \times 10^{-j} = 10^{-j}$.
            Therefore, as long as the decimal expansion terminates at digit $j$, we can increase $t_j$ by 1.
            This would mean that $s_j = t_j$, which would in turn hold true for all $k$.
            This, however, would have $a \in A$, which would contradict our supposition for contradiction.
        \end{case}

        \begin{case}{2}{$t_j - s_j \geq 1$}
            This will be pretty similar in execution to case 1.
            Take the above convergence equation.
            If $n > j$, then the maximum value of $t_k - s_k$ is $8$.
            Even if for all $k \geq n+1$, $t_k = 9$, it will always be the case that the magnitude of the sum up to and including $k = n$ will be greater than the sum after that point.
            \begin{gather}
                \scriptstyle
                \abs{\sum_{k=j}^{n} (s_k - t_k) \times 10^{-k} - \sum_{k=n+1}^{\infty} t_k \times 10^{-k}} = \abs{\sum_{k=n+1}^{\infty} t_k \times 10^{-k} - \sum_{k=j}^{n} (s_k - t_k) \times 10^{-k}}\\
                \scriptstyle
                \abs{\sum_{k=j}^{n} (s_k - t_k) \times 10^{-k} - \sum_{k=n+1}^{\infty} t_k \times 10^{-k}} = \abs{\sum_{k=n+1}^{\infty} t_k \times 10^{-k} + \sum_{k=j}^{n} (t_k - s_k) \times 10^{-k}}\\
                \abs{\sum_{k=n+1}^{\infty} t_k \times 10^{-k} + \sum_{k=j}^{n} (t_k - s_k) \times 10^{-k}} > 0\\
                \sum_{k=n+1}^{\infty} t_k \times 10^{-k} + \sum_{k=j}^{n} (t_k - s_k) \times 10^{-k} < \varepsilon
            \end{gather}

            Our sum of sums will always be greater than or equal to $10^{-j}$.
            \begin{gather}
                10^{-j} \leq \sum_{k=n+1}^{\infty} t_k \times 10^{-k} + \sum_{k=j}^{n} (t_k - s_k) \times 10^{-k} < \varepsilon
            \end{gather}

            This creates a contradiction, since $\varepsilon$ should be able to be less than $10^{-j}$.
            There is no edge case this time.
        \end{case}

        Ergo, our initial assumption is false, so $A$ is closed.
    \end{proof}
